CREATE TABLE jugadores(
id_jugador SMALLINT,
nombre varchar(50),
fecha_nac date,
demarcacion varchar(50),
internacional SMALLINT,
equipo varchar(50)
);

CREATE TABLE equipos(
id_equipo SMALLINT,
nombre varchar(50),
estadio varchar(50),
aforo SMALLINT,
ano_fundacion SMALLINT,
ciudad varchar(50)
);
